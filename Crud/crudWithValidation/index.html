    CrudWithReactiveValidation

    Steps of validation: Explanation of Each State:

    1.touched: Field has been focused and then blurred.
2.untouched: Field hasn't been touched yet.
3.dirty: Field value has been changed.
4.pristine: Field value has not changed.
5.valid: No validation errors.
6.invalid: Has validation errors.

    All validation types

    html
    <mat-form-field appearance="outline" class="mx-2">
  <mat-label>First Name</mat-label>
  <input matInput formControlName="firstNameFc" placeholder="Enter first name">

  <!-- Required error when touched or dirty -->
  <mat-error *ngIf="firstName?.hasError('required') && (firstName?.touched || firstName?.dirty)">
    First name is required
  </mat-error>

  <!-- State messages for demonstration -->
  <mat-hint *ngIf="firstName?.untouched">Status: untouched</mat-hint>
  <mat-hint *ngIf="firstName?.touched">Status: touched</mat-hint>
  <mat-hint *ngIf="firstName?.pristine">Status: pristine</mat-hint>
  <mat-hint *ngIf="firstName?.dirty">Status: dirty</mat-hint>
  <mat-hint *ngIf="firstName?.valid">First name is valid</mat-hint>
  <mat-hint *ngIf="firstName?.invalid">First name is invalid</mat-hint>
</mat-form-field>

    Crud -> With ReactiveForm Validation

    parent.ts
    import { Component, OnInit, ChangeDetectionStrategy } from '@angular/core';
import { FormGroup, FormControl, Validators } from '@angular/forms';
import { provideNativeDateAdapter } from '@angular/material/core';
import { Child } from '../child/child';

export interface DataFromParent {
  firstNameFc: string;
  lastNameFc: string;
  genderFc: string;
  dobFc: Date;
  cityFc: number;
  emailFc: string;
  ageFc: number;
  passwordFc: string;
  addressFc: string;
  appointmentTimeFc: string;
  cricketFc: boolean;
  footballFc: boolean;
  carromFc: boolean;
  skillsFc: string[];
  resumeFileName?: string;
  randomId: number;
}

@Component({
  selector: 'app-parent',
  standalone: true,
  imports: [
    Child,
    // Material modules here...
  ],
  changeDetection: ChangeDetectionStrategy.OnPush,
  providers: [provideNativeDateAdapter()],
  templateUrl: './parent.html',
  styleUrl: './parent.css'
})
export class Parent implements OnInit {
  myForm: FormGroup;
  city = [
    { cityId: 101, cityName: 'Hyderabad' },
    { cityId: 102, cityName: 'VijayaWada' },
    { cityId: 103, cityName: 'Maharastra' },
    { cityId: 104, cityName: 'Bengaloor' }
  ];
  skills: string[] = ['Angular', 'React', 'Vue', 'Node.js', 'Python'];
  data: DataFromParent[] = [];
  sendDataToChild: any;

  ngOnInit(): void {
    this.myForm = new FormGroup({
      randomId: new FormControl(Math.floor(Math.random() * 10000)),
      firstNameFc: new FormControl('', Validators.required),
      lastNameFc: new FormControl('', Validators.required),
      genderFc: new FormControl('', Validators.required),
      dobFc: new FormControl('', Validators.required),
      cityFc: new FormControl('', Validators.required),
      emailFc: new FormControl('', [Validators.required, Validators.email]),
      ageFc: new FormControl(null, Validators.required),
      passwordFc: new FormControl('', Validators.required),
      addressFc: new FormControl('', Validators.required),
      appointmentTimeFc: new FormControl('', Validators.required),
      cricketFc: new FormControl(false),
      footballFc: new FormControl(false),
      carromFc: new FormControl(false),
      skillsFc: new FormControl([], Validators.required),
      resumeFc: new FormControl(null)
    });
  }

  submitForm() {
    if (this.myForm.valid) {
      const formValue = this.myForm.value as DataFromParent;
      this.data.push(formValue);
      console.log(this.data);
      this.myForm.reset();
    }
  }

  receiveDataFromChild(data: DataFromParent[]): void {
    const item = data[0];
    this.myForm.patchValue(item);
  }

  onFileSelected(event: any) {
    const file = event.target.files[0];
    if (file) {
      this.myForm.patchValue({ resumeFc: file });
      const latest = this.data[this.data.length - 1];
      if (latest) latest.resumeFileName = file.name;
    }
  }

  onSelectall(event: any) {
    const all = event.checked;
    this.myForm.patchValue({
      skillsFc: all ? this.skills : []
    });
  }
}
