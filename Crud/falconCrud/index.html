    component.ts
    saveMspForm(): void {
  const id = this.mspGetData?.mspPatientInfoId;

  const mspForm = this.mspForm.value;
  // using form.value (no need to use get() every time)
  // direct ga form.value annam so no need to use 
  // this.form.get('')?.value directly use mspForm.firstName else 2ndu kuda use cheskovachu

  const mspObj = new Msp(); // create main object

  // for edit → if id vunte then edit , else 0 send chestam for save
  mspObj.mspPatientInfo.mspPatientInfoId = id ?? 0;

  // assigning values
  mspObj.mspPatientInfo.lastName = mspForm.lastName;
  mspObj.mspPatientInfo.firtName = mspForm.firtName;

  // Accident Details
  const accident = new MspAccidentDetail();
  accident.accidentType = mspForm.accidentType;
  accident.accidentLocation = mspForm.accidentLocation;

  // for edit → if id vunte then edit , else 0 send chestam for save
  accident.accidentID =
    this.mspGetData?.mspAccidentDetails?.[0]?.accidentID ?? 0;

  // 🔥 important: push into array
  mspObj.mspPatientInfo.mspAccidentDetails = [accident];

  // API call
  this.patientRegistrationService.postMSP(JSON.stringify(mspObj)).subscribe({
    next: (res) => {
      // success logic
    },
    error: (err) => {
      // error handling
    }
  });
}

    📦 Nested Model + Assignment
    model.ts
    export class MspAccidentDetail {
  accidentID: number = 0;
  accidentType: string = '';
  subAccidents: SubAccidentDetail[] = [];   // array
}

export class SubAccidentDetail {
  subId: number = 0;
  description: string = '';
  extraInfo: ExtraInfo = new ExtraInfo();   // object inside
}

export class ExtraInfo {
  severity: string = '';
  notes: string = '';
}
