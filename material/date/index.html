    TS

    startDate = new Date();
todayDate = new Date();

onFromDateChange(event: any): void {

  const fromDate = this.authorizationForm.get('approvedFrom')?.value;
  const toDate = this.authorizationForm.get('approvedTo')?.value;

  if (fromDate && toDate) {
    if (new Date(fromDate) > new Date(toDate)) {
      this.authorizationForm.get('approvedFrom')?.setErrors({ dateRangeInvalid: true });
    } else {
      this.authorizationForm.get('approvedFrom')?.setErrors(null);
    }
  }
}

    Explanation

    • todayDate = new Date() → Future dates are disabled using [max]="todayDate".
    • (dateChange) → Triggered when a date is selected from the calendar.
    • (dateInput) → Triggered when a date is typed manually.
    • onFromDateChange() → Validates that Approved From is not greater than Approved To.
    • setErrors({ dateRangeInvalid: true }) → Adds a custom validation error.
    • setErrors(null) → Clears the custom validation error.
    • mat-error with required → Shows required field validation message.
    • mat-error with dateRangeInvalid → Shows date range validation message.
    • [max]="todayDate" → User cannot select a future date.

    date range to use date range == if want only future dates then use [min]="startDate"

    How to use ngx-mask in Angular

    Angular ngx-mask Example

    Component Setup (TS File)

    import { NgxMaskDirective } from 'ngx-mask';

    HTML Usage

    <mat-form-field appearance="outline" class="example-full-width">
  <mat-label>Phone</mat-label>

  <input
    type="text"
    mask="000-000-0000"
    matInput
    placeholder="phone"
    formControlName="phone">
</mat-form-field>

    Installation

    npm install ngx-mask

    If you face version issues, install compatible version:

    npm install ngx-mask@20

    App Configuration (app.config.ts)

    import { provideNgxMask } from 'ngx-mask';

export const appConfig = {
  providers: [
    provideNgxMask()
  ]
};
