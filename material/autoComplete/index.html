    s-1:  imports: [ MatAutocompleteModule,MatOption,NgSelectModule,MatInputModule ] 
      drugList=[];
      druginputChange(event:Event){
      const inputValue = (event.target as HTMLInputElement).value; // what u enter in textBox that 
                                                              u will get into this variable
      this.historyService.getWsDrugSuggestions(inputValue,'DRUG','F').subscribe({
        next:(res)=>{
          this.drugList=res.body
        }
      })
    }
