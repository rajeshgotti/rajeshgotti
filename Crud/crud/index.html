@Component({
  selector: 'app-login',
  imports: [MatFormFieldModule,MatInputModule,ReactiveFormsModule,CommonModule],
  templateUrl: './login.html',
  styleUrl: './login.css'
})
export class Login {
  // isLogin:boolean=false
  loginForm:FormGroup

  constructor(private route:Router,private snack:MatSnackBar){
    this.loginForm=new FormGroup({
      usernameFc:new FormControl('',Validators.required),
      passwordFc:new FormControl('',Validators.required)

    })
  }

  //save loginForm
  Savelogin(){
    if(this.loginForm.valid){
      // this.isLogin=false;
      if(this.loginForm.get('usernameFc')?.value===this.loginForm.get('passwordFc')?.value){
        this.route.navigate(["/home"])
      }else{
        // alert('login failed!')
        this.snack.open('Invalid UserName Password!','Ok',{
          duration:2000
        })
      }
    }
  }
}

import { Component, Inject, PLATFORM_ID } from '@angular/core';
import { isPlatformBrowser, CommonModule } from '@angular/common';
import { AgGridModule } from 'ag-grid-angular';
import { ClientSideRowModelModule, ColDef, ModuleRegistry } from 'ag-grid-community';
import { MatDatepickerModule } from '@angular/material/datepicker';
import { FormArray, FormControl, FormGroup, ReactiveFormsModule, Validators } from '@angular/forms';
import { MatRadioModule } from '@angular/material/radio';
import { MatInputModule } from '@angular/material/input';

export class Home {
  columnDefs: ColDef[] = [];
  rowData: any;
  isBrowser: boolean;
  patientForm:FormGroup
  constructor(@Inject(PLATFORM_ID) platformId: Object,public dialog: MatDialog,private ser:Empservice) {
    this.isBrowser = isPlatformBrowser(platformId);
    this.patientForm=new FormGroup({
      fnameFc:new FormControl('',Validators.required),
      lnameFc:new FormControl(''),
      middleNameFc:new FormControl(''),
      dobFc:new FormControl(''),
      genderFc:new FormControl(''),
      nomineeFc:new FormControl('no'),
      formArrayFc:new FormArray([this.createNewAddress()])
    })
  }

  ngOnInit() {
    if (this.isBrowser) {
      this.getAggrid();
      // this.columnDefs = [
      //   { field: 'make', tooltipField: 'make' },
      //   { field: 'model', tooltipField: 'model' },
      //   { field: 'price', tooltipField: 'price' }
      // ];
      // this.rowData = [
      //   { make: 'Toyota', model: 'Celica', price: 35000 },
      //   { make: 'Ford', model: 'Mondeo', price: 32000 },
      //   { make: 'Porsche', model: 'Boxster', price: 72000 }
      // ];
    }
    this.getAllEmp();
  }

  getAggrid(){
    this.columnDefs=[
      {field:'empFirstNameDb',headerName:"FirstName"},
      {field:'empLastNameDb',headerName:'LastName'},
      {field:'empMiddleNameDb',headerName:'MiddleName'},
      {field:'empSalDb',headerName:'Salary'},
      {field:'genderDb',headerName:'Gender'},
      {field:'hobbiesDb',headerName:'Hobbies',
        cellRenderer:(params)=>{
          // return Array.isArray(params.value) ? params.value.join(', ') : params.value;
          return params.value? params.value.join(', '):params.value
        }
      },
      { 
        field: '', 
        headerName: 'Edit',
        cellRenderer: (params) => {
          const icon = document.createElement('i');
          icon.className = 'fa fa-pencil';  // Font Awesome pencil icon
          icon.style.cursor = 'pointer';
          icon.style.color = 'black';
          return icon;
        },
        onCellClicked: (params) => {
         this.editMethod(params.data)         
        }
      },
      {
        field: '',
        headerName: 'Delete',
        cellRenderer: () => {
          const icon = document.createElement('span');
          icon.className = 'material-icons';  // or 'fa fa-trash' if you use Font Awesome
          icon.style.cursor = 'pointer';
          icon.style.color = 'red';
          icon.innerText = 'delete'; // material delete icon
          return icon;
        },
        onCellClicked: (params) => {
          this.deleteMethod(params.data.id);  // pass row data to delete method
        }
      }



    ]
  }
  editMethod(params){
    this.patientForm.patchValue({
      fnameFc:params?.empFirstNameDb,
      lnameFc:params?.empLastNameDb,
      middleNameFc:params?.empMiddleNameDb,
      dobFc:params?.empDob,

    })
  }
  deleteMethod(id: number) {
    if (confirm('Are you sure you want to delete this employee?')) {
      this.ser.deleteEmp(id).subscribe({
        next: (res) => {
          console.log('Deleted successfully', res);
          this.getAllEmp(); // refresh grid
        },
        error: (err) => {
          console.error('Delete failed', err);
        }
      });
    }
  }

   get formArrayFc():FormArray{
      return this.patientForm.get('formArrayFc') as FormArray;
    }
    onNomineeChange(event:any){
      if(event?.value === 'yes'){
        this.dialog.open(Nominee,{
          height:'400px',
          width:'500px'
        })
      }else{
        console.log('no nominiee')
      }
    }
    addAddressBtn(){
      this.formArrayFc.push(this.createNewAddress())
    }
    removeAddressBtn(i){
    this.formArrayFc.removeAt(i)
    }
    createNewAddress():FormGroup{
  return new FormGroup({
    address1Fc:new FormControl(''),
    cityFc:new FormControl(''),
    stateFc:new FormControl('')
  })
    }

  getAllEmp(){
    this.ser.getEmployee().subscribe({
      next:(res)=>{
          // console.log(res,'getAllEmployees') 
          this.rowData = Array.isArray(res) ? res : [res];
          console.log(this.rowData,'rowData')
      },error(err) {

      },
    })
   } 
   savePatient(){
    let patientObj=new EmpModel();
    patientObj.empFirstNameDb=this.patientForm.get('fnameFc')?.value
    patientObj.empLastNameDb=this.patientForm.get('lnameFc')?.value
    patientObj.empMiddleNameDb=this.patientForm.get('middleNameFc')?.value
    patientObj.empDob=this.patientForm.get('dobFc')?.value
    patientObj.genderDb=this.patientForm.get('genderFc')?.value
    patientObj.addressesDb=this.addressMethod();
    this.ser.saveEmployee(patientObj).subscribe({
      next:(res)=>{
        // this.rowData=res
        this.getAllEmp();

        console.log(res,'saveResult')
      },error(err) {

      },
    })
    this.patientForm.reset();
   }
   addressMethod():any{
    const emptyArray:Address[]=[];

    let formArray=this.patientForm?.get('formArrayFc')?.value ?? [];
    formArray.forEach((ele,index)=>{
      const addressObj=new Address();
      addressObj.address1=ele?.address1Fc
      addressObj.city=ele?.cityFc
      addressObj.state=ele?.stateFc
      emptyArray.push(addressObj);
    })
   return emptyArray;

   }
}

    });

    const file = event.target.files[0];
    if (file) {
      this.myForm.patchValue({
        resumeFc: file,
      });

      const latest = this.data[this.data.length - 1];
      if (latest) {
        latest.resumeFileName = file.name;
      }
    }

      this.myForm.patchValue({ skillsFc: this.skills });

      this.myForm.patchValue({ skillsFc: [] });
    }
  }
