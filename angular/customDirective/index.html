    @HostListener

    What is @HostListener?

    @HostListener అంటే browser events ని listen చేయడానికి Angular లో ఉపయోగిస్తారు.

    ఉదాహరణకి:

    • scroll చేసినప్పుడు
    • resize చేసినప్పుడు
    • click చేసినప్పుడు
    • keyboard press చేసినప్పుడు

    Angular automatic గా method ని call చేస్తుంది.

    1. window:scroll

    @HostListener('window:scroll')
onScroll() {
  console.log('scrolling');
}

    Use:

    User page ని పైకి/కిందకి scroll చేసినప్పుడు trigger అవుతుంది.

    ఎక్కడ ఉపయోగిస్తారు?

    • Sticky Header
    • Back To Top button
    • Infinite Scroll
    • Lazy Loading
    • Scroll Position detect

    Example:

    YouTube/Facebook లో scroll చేస్తే కొత్త data load అవుతుంది.

    2. window:resize

    @HostListener('window:resize', ['$event'])
onResize(event: Event) {
  console.log('resized');
}

    Use:

    Browser window size change అయినప్పుడు.

    Example:

    • Desktop → Mobile view change
    • Responsive design
    • Sidebar hide/show

    Real Example:

    Screen చిన్నగా అయితే menu close చేయడం.

    3. document:click

    @HostListener('document:click', ['$event'])
onClick(event: MouseEvent) {
  console.log(event);
}

    Use:

    Page లో ఎక్కడైనా click detect చేయడానికి.

    Example:

    • Popup బయట click చేస్తే close
    • Dropdown close
    • Modal close

    4. keydown

    @HostListener('keydown')
onKeyDown() {
  console.log('key pressed');
}

    Use:

    Keyboard key press detect చేయడానికి.

    Example:

    • Shortcut keys
    • Session timeout reset
    • Validation

    5. keyup

    @HostListener('keyup', ['$event'])
onKeyUp(event: KeyboardEvent) {
  console.log(event);
}

    Use:

    User typing పూర్తయ్యాక detect చేయడానికి.

    Example:

    • Search filter
    • Convert uppercase
    • Input validation

    Note:

    మీ uppercase directive ఇదే use చేస్తుంది.

    6. mouseenter

    @HostListener('mouseenter')
onMouseEnter() {
}

    Use:

    Mouse element మీదికి వచ్చినప్పుడు.

    Example:

    • Tooltip show
    • Hover effect
    • Highlight

    7. mouseleave

    @HostListener('mouseleave')
onMouseLeave() {
}

    Use:

    Mouse బయటికి వెళ్లినప్పుడు.

    Example:

    • Tooltip hide
    • Remove highlight

    8. window:unload

    @HostListener('window:unload')
onUnload() {
}

    Use:

    Browser/tab close అయినప్పుడు.

    Example:

    • Logout
    • Save session
    • Stop SignalR connection

    Simple Meaning

    Event Meaning
    scroll page scroll అయ్యింది
    resize screen size మారింది
    click mouse click
    keydown key press
    keyup key release
    mouseenter mouse వచ్చింది
    mouseleave mouse వెళ్లింది
    unload page close అయ్యింది

    Without @HostListener

}

    1. Uppercase Directive

    TypeScript

    import { Directive, HostListener, ElementRef } from '@angular/core';

@Directive({
  selector: '[appUppercase]'
})
export class UppercaseDirective {

  constructor(private el: ElementRef) {}

  @HostListener('input')
  onInput() {
    this.el.nativeElement.value =  this.el.nativeElement.value.toUpperCase();
  }
}

    Custom Directive - Allow Only 2 Characters

    Directive

    import { Directive, ElementRef, HostListener } from '@angular/core';

@Directive({
  selector: '[appAllow2chars]'
})
export class Allow2chars {

  constructor(private el: ElementRef<HTMLInputElement>) {}

  @HostListener('input')
  onInput() {
    let res = this.el.nativeElement.value.toUpperCase();

    res = res.slice(0, 2);

    this.el.nativeElement.value = res;
  }
}

    2. Only Number Directive

    TypeScript

    import { Directive, HostListener } from '@angular/core';

@Directive({
  selector: '[appOnlyNumber]'
})
export class OnlyNumberDirective {

  @HostListener('keypress', ['$event'])
  onKeyPress(event: KeyboardEvent) {

    const charCode = event.key.charCodeAt(0);

    if (charCode < 48 || charCode > 57) {
      event.preventDefault();
    }
  }
}

    3. Highlight Directive

    TypeScript

    import { Directive, HostListener,Renderer2, ElementRef } from '@angular/core';

@Directive({
  selector: '[appHighlight]'
})
export class HighlightDirective {

  constructor(
    private el: ElementRef,
    private renderer: Renderer2
  ) {}

  @HostListener('mouseenter')
  onEnter() {
    this.renderer.setStyle(
      this.el.nativeElement,
      'backgroundColor',
      'yellow'
    );
  }

  @HostListener('mouseleave')
  onLeave() {
    this.renderer.removeStyle(
       this.el.nativeElement,
      'backgroundColor'
    );
  }
}

    HTML

    <div appHighlight> Hover me</div>

    7. Debounce Search Directive

    TypeScript

    import {Directive,HostListener} from '@angular/core';
import { Subject } from 'rxjs';
import { debounceTime } from 'rxjs/operators';

@Directive({
  selector: '[appDebounceSearch]'
})
export class DebounceSearchDirective {

  private search = new Subject<string>();

  constructor() {
    this.search.pipe(debounceTime(500)).subscribe(value => {
        console.log('API Call:', value);
      });
  }

  @HostListener('input', ['$event'])
  onInput(event: any) {
    this.search.next(event.target.value);
  }
}

    HTML

    <input type="text"  appDebounceSearch placeholder="Search">
    8. Date Format Directive

    TypeScript

    import { Directive, HostListener,ElementRef} from '@angular/core';

@Directive({
  selector: '[appDateFormat]'
})
export class DateFormatDirective {

  constructor(private el: ElementRef) {}

  @HostListener('input')
  onInput() {

    let value = this.el.nativeElement.value;

    value = value.replace(/\D/g, '');

    if (value.length >= 2) {
      value = value.substring(0, 2) + '/' + value.substring(2);
    }

    if (value.length >= 5) {
      value =  value.substring(0, 5) +  '/' + value.substring(5, 9);
    }

    this.el.nativeElement.value = value;
  }
}

    HTML

    <input type="text" appDateFormat placeholder="MM/dd/YYYY">

    9. Permission Directive

    TypeScript

    import { Directive,Input,TemplateRef, ViewContainerRef} from '@angular/core';

@Directive({
  selector: '[appHasPermission]'
})
export class HasPermissionDirective {

  @Input()
  set appHasPermission(role: string) {

    const userRole = 'ADMIN';

    if (role === userRole) {
      this.vc.createEmbeddedView(this.tpl);
    } else {
      this.vc.clear();
    }
  }

  constructor(private tpl: TemplateRef<any>, private vc: ViewContainerRef) {}
}

    HTML

    <button *appHasPermission="'ADMIN'">
  Delete User
</button>
    10. Lazy Load Image Directive

    TypeScript

    import { Directive, ElementRef, HostListener } from '@angular/core';

@Directive({
  selector: '[appLazyLoad]'
})
export class LazyLoadDirective {

  constructor(private el: ElementRef) {}

  @HostListener('window:scroll')
  onScroll() {
    const rect = this.el.nativeElement.getBoundingClientRect();

    if (rect.top < window.innerHeight) {
      const src =this.el.nativeElement.getAttribute('data-src');
      if (src) {
        this.el.nativeElement.src = src;
      }
    }
  }
}

    HTML

    <img appLazyLoad
  data-src="assets/images/demo.jpg"
  alt="Demo Image">
