    NGXS Store - Step by Step (Telugu)

    1. Store అంటే ఏమిటి?

    Store అంటే ఒక Central Data Box (Global Memory).

Application లో ఎక్కడైనా, ఏ Component నుండైనా
Data ని Save, Read, Update చేయవచ్చు.

Falcon Project లో Store కోసం NGXS Library ఉపయోగిస్తారు.

Data SessionStorage లో Persist అవుతుంది.
(Page Refresh అయినా Data పోదు.)

    patientState.ts 
    import { Injectable } from '@angular/core';
import {
  Action,
  Selector,
  State,
  StateContext
} from '@ngxs/store';

import {
  ClearPatientName,
  SetPatientName
} from './patientAction';

export interface PatientStateModel {
  patientName: string | null;
}

@State<PatientStateModel>({
  name: 'PatientState',
  defaults: {
    patientName: null,
  },
})

@Injectable()
export class PatientStateClass {

  @Selector()
  static getPatientName(
    state: PatientStateModel
  ): string | null {
    return state.patientName;
  }

  @Action(SetPatientName)
  setPatientName(
    ctx: StateContext<PatientStateModel>,
    action: SetPatientName
  ) {
    ctx.patchState({
      patientName: action.payload
    });
  }

  @Action(ClearPatientName)
  clearPatientName(
    ctx: StateContext<PatientStateModel>
  ) {
    ctx.patchState({
      patientName: null
    });
  }

}

// STEP 3: STATE (erdashboard.state.ts)
import { Injectable } from '@angular/core';
import { State, Action, StateContext, Selector } from '@ngxs/store';
import {
  clearPatientIdVisitIdAction,
  storePatientIdVisitIdAction
} from './erdashboard.action';

export class storePatientIdVisitIdModel {
  patientId: any;
  visitId: any;
  gender: string;
  patientModel: PatientInformation;
}

@State<storePatientIdVisitIdModel>({
  name: 'patientIdVisitId',
  defaults: {
    patientId: 0,
    visitId: 0,
    gender: '',
    patientModel: null,
  }
})
@Injectable()
export class storePatientIdVisitIdState {

  @Selector()
  static getPatientIdVisitId(state: storePatientIdVisitIdModel) {
    return state;
  }

  @Action(storePatientIdVisitIdAction)
  storePatientIdVisitIdAction(
    { patchState }: StateContext<storePatientIdVisitIdModel>,
    { payload }: storePatientIdVisitIdAction
  ) {
    const { patientModel } = payload || {};

    if (patientModel) {
      patientModel.patientType =
        patientModel.patientType ?? patientModel.patientVisitType;
    }

    patchState({
      patientId: payload.patientId,
      visitId: payload.visitId,
      gender: payload.gender,
      patientModel: patientModel,
    });
  }

  @Action(clearPatientIdVisitIdAction)
  clearPatientIdVisitIdAction(
    { patchState }: StateContext<storePatientIdVisitIdModel>
  ) {
    patchState({
      patientId: 0,
      visitId: 0,
      gender: '',
      patientModel: null,
    });
  }
}

    Ngrx operators

    1. filter – Allow Only Values That Meet a Condition

import { CommonModule } from '@angular/common';
import { Component, OnInit } from '@angular/core';
import { of } from 'rxjs';
import { filter } from 'rxjs/operators';

@Component({
  selector: 'app-demo',
  standalone: true,
  imports: [CommonModule],
  templateUrl: './demo.component.html',
  styleUrls: ['./demo.component.scss']
})
export class DemoComponent implements OnInit {
  myData: number[] = [];

  ngOnInit(): void {
    of(1, 2, 3, 4, 5).pipe(
      filter(x => x % 2 === 0)
    ).subscribe(res => {
      this.myData.push(res); //outPut: 2 4
    });
  }
}

----------Take------
2. take – Limit to First N Values
The take operator emits only the first N values, then completes.
import { interval } from 'rxjs';
import { take } from 'rxjs/operators';

interval(1000).pipe(
  take(3)
).subscribe(console.log); //output: 0 1 2

-------First-----
3. first – Emit Only the First Value (Optionally Matching a Condition)
The first operator emits only the first value from the stream and completes.

import { of } from 'rxjs';
import { first } from 'rxjs/operators';

of(1, 3, 7, 9).pipe(
  first(x => x > 5)
).subscribe(console.log);// Output: 7
---------Last-------
4. last – Emit Only the Last Value (Optionally Matching a Condition)
The last operator waits for the stream to complete and emits the last value (optionally matching a predicate).

import { of } from 'rxjs';
import { last } from 'rxjs/operators';

of(1, 4, 6, 9).pipe(
  last(x => x % 2 === 0)
).subscribe(console.log); // Output: 6

-----------skip------

5. skip – Ignore First N Values
The skip operator skips the first N emitted values, then emits the rest.

import { of } from 'rxjs';
import { skip } from 'rxjs/operators';

of(10, 20, 30, 40).pipe(
  skip(2)
).subscribe(console.log); //30  40
