    import { Component, OnInit } from '@angular/core';
import { AgGridModule } from 'ag-grid-angular';
import { ClientSideRowModelModule, ColDef, ModuleRegistry } from 'ag-grid-community';

// ✅ Register the module
ModuleRegistry.registerModules([ClientSideRowModelModule]);

@Component({
  selector: 'app-aggrid',
  standalone: true,
  imports: [AgGridModule],
  templateUrl: './aggrid.html',
  styleUrl: './aggrid.css'
})
export class Aggrid implements OnInit {
  columnDefs: ColDef[] = [];
  rowData: any[] = [];
  ngOnInit() {
    this.columnDefs = [
      { field: 'make', tooltipField: 'make' },
      { field: 'model', tooltipField: 'model' },
      { field: 'price', tooltipField: 'price' }
    ];
    this.rowData = [
      { make: 'Toyota', model: 'Celica', price: 35000 },
      { make: 'Ford', model: 'Mondeo', price: 32000 },
      { make: 'Porsche', model: 'Boxster', price: 72000 }
    ];
  }
}

    valueFormater

    valueFormatter:The valueFormatter is used in AG Grid to format the raw value from your data 
before displaying it in the cell.
  - It does not change the actual data — only how it looks in the UI.
  -Imagine your data contains a raw timestamp, 
Example: {
            lastModifiedDate: "2025-08-08T14:23:45.000Z"
         } 
 You probably don’t want to show that long timestamp You want to show something 
  friendly like:08/08/2025 14:23 then use valueFormater
{ 
  field: 'lastModifiedDate',
  headerName: 'Status',
  headerTooltip: "lastModifiedDate",
  valueFormatter: function (params) {
    return moment(params.value).format('MM/DD/YYYY HH:mm');
  }
}

    valueGetter (to combine 2 fields we use this)

    ->valueGetter:Add two fields? (e.g., price + tax) => Combine fields? (e.g., firstName + lastName)
   =>Use a conditional? (e.g., if (stock > 0) 'In Stock' else 'Out of Stock') 
Example:{
          headerName: 'Full Name',
          valueGetter: (params) => {
          return params.data.firstName + ' ' + params.data.lastName;
         }
        }
Example:2{
          headerName: 'Full Name',
          valueGetter: params => ${params.data.firstName} ${params.data.lastName}
          }
          Calculate a new field:{
              headerName: 'Total Price',
              valueGetter: params => params.data.price + params.data.tax
            }
Example 3: Conditional value
          {
            headerName: 'Availability',
            valueGetter: params => params.data.stock > 0 ? 'In Stock' : 'Out of Stock'
          }

    rowClassRules

    ts.file
      rowClassRules is just like cellClassRules — but it applies CSS classes to entire rows,
  instead of individual cells.
Purpose:
To dynamically style full rows based on conditions in your data.
 Basic Syntax:
 rowClassRules: {
  'row-error': params => params.data.status === 'error',
  'row-success': params => params.data.status === 'success'
}

----html----
<ag-grid-angular
  class="ag-theme-alpine"
  [rowData]="rowData"
  [columnDefs]="columnDefs"
  [rowClassRules]="rowClassRules">//rowClassRules
</ag-grid-angular>

---ts.file----
rowClassRules = {
  'row-success': params => params.data.status === 'success',
  'row-error': params => params.data.status === 'error',
  'row-pending': params => params.data.status === 'pending'
};

--css.file--
.row-success {
  background-color: #e0fce0 !important;
}

.row-error {
  background-color: #fde0e0 !important;
}

.row-pending {
  background-color: #fff8e0 !important;
}
