    ⚙️ Component (Full TS File)

    import { CommonModule } from '@angular/common';
import { Component, OnInit } from '@angular/core';
import { FormArray, FormControl, FormGroup, ReactiveFormsModule, Validators } from '@angular/forms';
import { provideNativeDateAdapter } from '@angular/material/core';
import { MatDatepickerModule } from '@angular/material/datepicker';
import { MatFormFieldModule } from '@angular/material/form-field';
import { MatInputModule } from '@angular/material/input';
import { MatRadioModule } from '@angular/material/radio';
import { MatDialog } from '@angular/material/dialog';

import { Nominee } from '../nominee/nominee';
import { Empservice } from '../../Service/empservice';
import { Address, EmpModel } from '../../../EmployeeModel';

@Component({
  selector: 'app-patient',
  standalone: true,
  imports: [
    CommonModule,
    ReactiveFormsModule,
    MatFormFieldModule,
    MatInputModule,
    MatRadioModule,
    MatDatepickerModule
  ],
  providers: [provideNativeDateAdapter()],
  templateUrl: './patient.html',
  styleUrl: './patient.css'
})
export class Patient implements OnInit {

  patientForm: FormGroup;

  constructor(
    public dialog: MatDialog,
    private ser: Empservice
  ) {
    this.patientForm = new FormGroup({
      fnameFc: new FormControl('', Validators.required),
      lnameFc: new FormControl(''),
      middleNameFc: new FormControl(''),
      dobFc: new FormControl(''),
      genderFc: new FormControl(''),
      nomineeFc: new FormControl('no'),
      formArrayFc: new FormArray([this.createNewAddress()])
    });
  }

  ngOnInit(): void {
    this.getAllEmp();
  }

  get formArrayFc(): FormArray {
    return this.patientForm.get('formArrayFc') as FormArray;
  }

  onNomineeChange(event: any) {
    if (event?.value === 'yes') {
      this.dialog.open(Nominee, {
        height: '400px',
        width: '500px'
      });
    } else {
      console.log('no nominee');
    }
  }

  addAddressBtn() {
    this.formArrayFc.push(this.createNewAddress());
  }

  removeAddressBtn(i: number) {
    this.formArrayFc.removeAt(i);
  }

  createNewAddress(): FormGroup {
    return new FormGroup({
      address1Fc: new FormControl(''),
      cityFc: new FormControl(''),
      stateFc: new FormControl('')
    });
  }

  getAllEmp() {
    this.ser.getEmployee().subscribe({
      next: (res) => {
        console.log(res, 'getAllEmployees');
      },
      error: (err) => {
        console.error(err);
      }
    });
  }

  savePatient() {
    let patientObj = new EmpModel();

    patientObj.empFirstNameDb = this.patientForm.get('fnameFc')?.value;
    patientObj.empLastNameDb = this.patientForm.get('lnameFc')?.value;
    patientObj.empMiddleNameDb = this.patientForm.get('middleNameFc')?.value;
    patientObj.empDob = this.patientForm.get('dobFc')?.value;
    patientObj.genderDb = this.patientForm.get('genderFc')?.value;

    patientObj.addressesDb = this.addressMethod();

    console.log('Final Payload:', patientObj);
  }

  addressMethod(): Address[] {
    const emptyArray: Address[] = [];

    let formArray = this.patientForm.get('formArrayFc')?.value ?? [];

    formArray.forEach((ele: any) => {
      const addressObj = new Address();

      addressObj.address1 = ele.address1Fc;
      addressObj.city = ele.cityFc;
      addressObj.state = ele.stateFc;

      emptyArray.push(addressObj);
    });

    return emptyArray;
  }
}
