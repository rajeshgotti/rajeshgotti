    Example:1 basic

    Note:import MatExpansionModule it is enought no need to import MatExpansionHeader,title etc

    ts.file
    import { MatExpansionModule } from '@angular/material/expansion';

    <mat-expansion-panel  [expanded]="true"> // true iste then default ga open vuntundi false iste close lo vuntundi

  <!-- Expansion Panel Header START -->
  <mat-expansion-panel-header>

        <!-- Panel Title -->
        <mat-panel-title>
          <strong>Drug Details</strong>
        </mat-panel-title>

        <!-- Panel Description -->
        <mat-panel-description>
          <p>This is optional</p>
        </mat-panel-description>

  </mat-expansion-panel-header>
  <!-- Expansion Panel Header END -->

  <!-- Main Content -->
  <p>Here main content we write</p>
  <!-- Action Row -->
  <mat-action-row>
    <button mat-button> Next </button>
  </mat-action-row>

</mat-expansion-panel>

    Example:2 normal

    The is a wrapper component in Angular Material used to group and manage multiple components.

    html.file
    <mat-accordion>
    <!-- pannel 1-->
  <mat-expansion-panel> ... </mat-expansion-panel>
   <!-- pannel 2-->
  <mat-expansion-panel> ... </mat-expansion-panel>
</mat-accordion>
    html.file
    <!-- mat-accordion is used to group multiple expansion panels-->

<mat-accordion multi="true">

    <mat-expansion-panel> //mat-expansion-panel START
      <mat-expansion-panel-header>
        <mat-panel-title>
          Panel 1
        </mat-panel-title>
      </mat-expansion-panel-header>

      <p>Content 1</p>
    </mat-expansion-panel> //mat-expansion-panel END


    <mat-expansion-panel>  //START
      <mat-expansion-panel-header>
        <mat-panel-title>
          Panel 2
        </mat-panel-title>
      </mat-expansion-panel-header>

      <p>Content 2</p>
    </mat-expansion-panel> //END

</mat-accordion>
    ts.file
    multi="true"
→ Allows multiple panels to stay open at once

multi="false" (default)
→ Only one panel stays open
→ Other panels automatically close

})

export class AddDrugComponent {

}

    note: 1: mat-accordion multiple="true" -> Users can open and close any panel independently.
         if multiple="false" iste then  -> then user oka sari okka panel ni matarame open ceyali inkoti open 
         cheste edi close avtundi 
      2:  <mat-expansion-panel expanded="true"> 
            expanded="true" -> true iste then expansionPanel default ga open aiyi vuntundi 
            expanded="false" -> false iste then panel default ga close ai vuntundi
      3:  hideToggle="true" -> true iste ha downArrow icon anedi disable aipotundi but
          still if u click on headerPart it will work
