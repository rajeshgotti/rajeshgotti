    import { Component, OnInit } from '@angular/core';
import { AbstractControl, FormArray, FormBuilder, FormGroup, Validators } from '@angular/forms';
import { CommonModule } from '@angular/common';
import { ReactiveFormsModule } from '@angular/forms';

@Component({
  selector: 'app-form-array',
  standalone: true,
  imports: [CommonModule, ReactiveFormsModule],
  templateUrl: './form-array.component.html',
  styleUrls: ['./form-array.component.css']
})
export class FormArrayComponent implements OnInit {
  formArray!: FormGroup;
  submittedData: any[] = [];

  isEditing = false;
  editingIndex: number | null = null;

  constructor(private fb: FormBuilder) {}

  ngOnInit(): void {
    this.initForm();
  }

  // Initialize form
  initForm(): void {
    this.formArray = this.fb.group({
      userDetails: this.fb.array([this.createUser()])
    });
  }

  // Create a user form group
  createUser(): FormGroup {
    return this.fb.group({
      firstname: ['', Validators.required],
      lastname: ['', Validators.required],
      gender: [''],
      textme: [''],
      yes: [false],
      no: [false]
    });
  }

  // Getter for userDetails
  get userDetails(): FormArray {
    return this.formArray.get('userDetails') as FormArray;
  }

  // Add new user
  addUser(): void {
    this.userDetails.push(this.createUser());
  }

  // Remove user
  removeUser(index: number): void {
    this.userDetails.removeAt(index);
  }

  // Submit form
onSubmit(): void {
  if (this.isEditing && this.editingIndex !== null) {
    // Update existing record
    this.submittedData[this.editingIndex] = this.extractFormData(this.userDetails.at(0));
    this.isEditing = false;
    this.editingIndex = null;
  } else {
    // Submit all form entries
    this.submittedData = this.userDetails.controls.map(control => this.extractFormData(control));
  }

  console.log('Submitted:', this.submittedData);

  // ✅ Reset form after submission
  this.userDetails.clear();
  this.addUser();
  this.formArray.markAsPristine();
  this.formArray.markAsUntouched();
}


  // Extract data from a FormGroup
  private extractFormData(control: AbstractControl): any {
    return {
      firstname: control.get('firstname')?.value,
      lastname: control.get('lastname')?.value,
      gender: control.get('gender')?.value,
      checkboxYes: control.get('yes')?.value,
      checkboxNo: control.get('no')?.value,
      message: control.get('textme')?.value
    };
  }

  // Edit existing user
  editUser(index: number): void {
    this.isEditing = true;
    this.editingIndex = index;

    const selectedData = [this.submittedData[index]];

    this.userDetails.clear(); // clear existing form
    this.patchValue(selectedData);
  }

  // Patch form values
  patchValue(getdata: any[]): void {
    getdata.forEach((data, index) => {
      const formGroup = this.createUser();
      formGroup.patchValue({
        firstname: data.firstname,
        lastname: data.lastname,
        gender: data.gender,
        yes: data.checkboxYes,
        no: data.checkboxNo,
        textme: data.message
      });
      this.userDetails.push(formGroup);
    });
  }

  // Delete submitted user
  deleteUser(index: number): void {
    this.submittedData.splice(index, 1);

  }
}

ngOnInit(): void {
 if(this.drugLibraryId){  // if ID present then(editOprtion) this method will execute else save
      this.pharmacyDrugInfo();

}

pharmacyDrugInfo(): void {
  this.pharmacyService.getpharmacyDrugDetails(this.type, this.drugLibraryId, '')
    .pipe(takeUntil(this.onDestroy$))
    .subscribe(res => {
      this.pharmacyDrugInfoData = res.body[0];
      this.SearchDrugId = this.pharmacyDrugInfoData?.drugid?.toString();

      if (this.selectedDrugType !== 'Vaccine') {
        this.pharmacyForm.reset();
        this.pharmacyForm.patchValue({
          // Unit Dose Patch
          drugName: this.pharmacyDrugInfoData?.genericname ?? null,
          barndName: this.pharmacyDrugInfoData?.drugname ?? null,
        });
      }

      if (this.selectedDrugType === 'Vaccine') {
        this.vaccinationFormPatch();
      }
    });
}

vaccinationFormPatch(): void {
  this.getSelectedVaccinationData(this.pharmacyDrugInfoData?.vaccineLotsList);

  this.vaccinationForm.patchValue({
    vaccineName: this.pharmacyDrugInfoData?.genericname || null,
    doseStrength: this.pharmacyDrugInfoData?.med_strength || null,
    doseUnit: this.pharmacyDrugInfoData?.med_unit || null,
    routeName: this.routes?.find(
      ele => ele.meD_ROUTE_ID == this.pharmacyDrugInfoData?.route
    ) ?? null,
    rxNormCode: this.pharmacyDrugInfoData?.rxNormCode || null,
    ndc: this.pharmacyDrugInfoData?.ndc || null,
    note: this.pharmacyDrugInfoData?.note || null,
  });

  const lotsFormArray = this.vaccineLotsList;

  while (lotsFormArray.length) {
    lotsFormArray.removeAt(0);
  }

  if (this.pharmacyDrugInfoData?.vaccineLotsList?.length) {
    this.pharmacyDrugInfoData.vaccineLotsList.forEach((lot: VaccinationList) => {
      lotsFormArray.push(this.createLot(lot));
    });
  }
}

private createLot(lot: VaccinationList): FormGroup {
  return this.fb.group({
    vaccineLotID: [lot?.vaccineLotID || null],
    lotNumber: [lot?.lotNumber || null],
    expireDate: [lot?.expireDate ? new Date(lot.expireDate) : null],
  });
}

 saveCareTeamForm() {
 this.addCareTeamForm.markAllAsTouched();
    if (this.addCareTeamForm.valid) {
      const formValue = this.addCareTeamForm.value;

      const careTeamObj = new CareTeam(); // creating obj 
      careTeamObj.careTeamNameID = this.data?.careTeamData?.careTeamNameID ?? 0
      careTeamObj.careTeamName = formValue?.careTeamName ?? '';
      careTeamObj.categoryID = formValue?.Catagory ?? null;
      careTeamObj.status = formValue?.status?.toString() ?? '';
      careTeamObj.reason = formValue?.reason ?? '';
      careTeamObj.fromDate =formValue?.fromDate ?? new Date()
      careTeamObj.toDate = formValue?.toDate ?? new Date()
      careTeamObj.enteredBy = this.userId;

         <!-- members formArray starts form here with loop -->
      careTeamObj.members = (formValue?.careTeam  || []).map((user: any) => {

        const contactObj = new CareTeamMember(); //creating obj for CareTeamMember
        contactObj.careTeamMemberID = user.careTeamMemberID ?? 0;
        contactObj.userID =  user.username;
        contactObj.userTypeID = user?.role?.toLowerCase()?.includes('nurs') ? 2 : 1;
        contactObj.userRoleId = 0
        contactObj.enteredBy = this.userId;
        contactObj.subRoleCode = this.roleData.find((item) => item?.description === user?.role)?.code ?? '';
        contactObj.subRoleDesc= user?.role
        contactObj.enteredDate = new Date().toISOString();
        return contactObj;
      });

        careTeamObj.contactInfos = (formValue?.creatcontactInfo || []).map((contact: any) => {

        const contactObj = new ContactInfo();   // creating obj for ContactInfo
        contactObj.contactID = contact.contactID ?? 0;
        contactObj.userTypeID = 5
        contactObj.use = contact.useType ?? '';
        contactObj.systemID = this.system.find((item) => item.display === contact.system)?.contactPointID ?? 0;
        contactObj.value = contact.value ?? '';
        contactObj.rank = contact.rank=="" ? 0 : contact.rank;
        return contactObj;
      });
      console.log('✅ Care Team Object to send:', careTeamObj);
      this._CareTeamService.postCareTeam(careTeamObj).subscribe({
        next: (res) => {
          this.clearCareTeamForm();
          this.snackBar.showSnackBar('Care Team Saved successfully!');
          this.dialogRef.close('success');
        },
        error: (err) => {
          console.error('Error submitting:', err);
        }
      });
    } else {
      this.addCareTeamForm.markAllAsTouched();
    }

}
