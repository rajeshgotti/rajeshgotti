BEGIN

    STEP 2: Create Patients Table

    IF NOT EXISTS (SELECT * FROM sys.tables WHERE name = 'Patients')
BEGIN
    CREATE TABLE Patients
    (
        PatientId     INT IDENTITY(1,1) PRIMARY KEY,
        PatientFName  VARCHAR(50)  NOT NULL,
        MiddleName    VARCHAR(50)  NULL,
        LastName      VARCHAR(50)  NOT NULL,
        DOB           DATETIME     NOT NULL,
        IsPrivate     BIT          NOT NULL DEFAULT 0,
        Hobbies       VARCHAR(500) NULL,
        SelectState   VARCHAR(50)  NULL
    );
END
GO

    🔹 Stored Procedure 1: sp_GetAllPatients

    New Query window → paste → Execute

    USE emrdb;
GO

IF OBJECT_ID('sp_GetAllPatients', 'P') IS NOT NULL
    DROP PROCEDURE sp_GetAllPatients;
GO

CREATE PROCEDURE sp_GetAllPatients
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        PatientId,
        PatientFName,
        MiddleName,
        LastName,
        DOB,
        IsPrivate,
        Hobbies,
        SelectState
    FROM Patients
    ORDER BY PatientId DESC;

END;
GO

    🔹 Stored Procedure 2: sp_GetPatientById

    USE emrdb;
GO

IF OBJECT_ID('sp_GetPatientById', 'P') IS NOT NULL
    DROP PROCEDURE sp_GetPatientById;
GO

CREATE PROCEDURE sp_GetPatientById
    @PatientId INT
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        PatientId,
        PatientFName,
        MiddleName,
        LastName,
        DOB,
        IsPrivate,
        Hobbies,
        SelectState
    FROM Patients
    WHERE PatientId = @PatientId;

END;
GO

    🔹 Stored Procedure 3: sp_AddPatient (Insert + Update)

    💡 @PatientId = 0 → INSERT | @PatientId > 0 → UPDATE

    Same as your Mobile SP pattern.

    USE emrdb;
GO

IF OBJECT_ID('sp_AddPatient', 'P') IS NOT NULL
    DROP PROCEDURE sp_AddPatient;
GO

CREATE PROCEDURE sp_AddPatient
    @PatientId    INT,
    @PatientFName VARCHAR(50),
    @MiddleName   VARCHAR(50),
    @LastName     VARCHAR(50),
    @DOB          DATETIME,
    @IsPrivate    BIT,
    @Hobbies      VARCHAR(500),
    @SelectState  VARCHAR(50)
AS
BEGIN
    SET NOCOUNT ON;

    IF (@PatientId = 0)
    BEGIN
        -- INSERT new patient
        INSERT INTO Patients
        (
            PatientFName,
            MiddleName,
            LastName,
            DOB,
            IsPrivate,
            Hobbies,
            SelectState
        )
        VALUES
        (
            @PatientFName,
            @MiddleName,
            @LastName,
            @DOB,
            @IsPrivate,
            @Hobbies,
            @SelectState
        );
    END
    ELSE
    BEGIN
        -- UPDATE existing patient
        UPDATE Patients
        SET
            PatientFName = @PatientFName,
            MiddleName   = @MiddleName,
            LastName     = @LastName,
            DOB          = @DOB,
            IsPrivate    = @IsPrivate,
            Hobbies      = @Hobbies,
            SelectState  = @SelectState
        WHERE PatientId = @PatientId;
    END

END;
GO

    🔹 Stored Procedure 4: sp_DeletePatient

    USE emrdb;
GO

IF OBJECT_ID('sp_DeletePatient', 'P') IS NOT NULL
    DROP PROCEDURE sp_DeletePatient;
GO

CREATE PROCEDURE sp_DeletePatient
    @PatientId INT
AS
BEGIN
    SET NOCOUNT ON;

    DELETE FROM Patients
    WHERE PatientId = @PatientId;

END;
GO

    💡 Key Points

    Feature Purpose
    IF OBJECT_ID(...) DROP SP already ఉంటే drop చేసి re-create; re-run safe
    SET NOCOUNT ON Extra "rows affected" message hide చేస్తుంది
    IDENTITY(1,1) PatientId auto-increment — 1, 2, 3...
    PatientId = 0 for INSERT 0 pass చేస్తే INSERT, existing ID ఉంటే UPDATE
    DATETIME for DOB Time component కూడా store అవుతుంది
    DATE Only date కావాలంటే DATE use చేయండి

    ⚠️ Common Errors & Fixes

    Error Reason Fix
    Cannot find database emrdb Database create అవలేదు Step 1 script run చేయండి
    Invalid object name Patients Table create అవలేదు Step 2 script run చేయండి
    Procedure already exists SP already exists DROP PROCEDURE already handled
    Cannot insert NULL into column PatientFName Required field missing Angular form validation add చేయండి

    STEP 8: Repository Implementation

    Right-click on Repositories folder → Add → Class → Name: PatientRepository.cs

    using LoginPatient.Data;
using LoginPatient.Models;
using Microsoft.EntityFrameworkCore;

namespace LoginPatient.Repositories
{
    public class PatientRepository : IPatientRepository
    {
        private readonly PatientDbContext _context;

        public PatientRepository(PatientDbContext context)
        {
            _context = context;
        }

        public async Task<List<PatientModel>> GetAllPatientsAsync()
        {
            return await _context.Patients
                .FromSqlRaw("EXEC sp_GetAllPatients")
                .ToListAsync();
        }

        public async Task<PatientModel?> GetPatientByIdAsync(int patientId)
        {
            var result = await _context.Patients
                .FromSqlInterpolated($"EXEC sp_GetPatientById @PatientId={patientId}")
                .ToListAsync();

            return result.FirstOrDefault();
        }

        public async Task<int> AddPatientAsync(PatientModel patient)
        {
            var result = await _context.Database.ExecuteSqlInterpolatedAsync(
                $@"EXEC sp_AddPatient 
                    @PatientId={patient.PatientId},
                    @PatientFName={patient.PatientFName},
                    @MiddleName={patient.MiddleName},
                    @LastName={patient.LastName},
                    @DOB={patient.DOB},
                    @IsPrivate={patient.IsPrivate},
                    @Hobbies={patient.Hobbies},
                    @SelectState={patient.SelectState}"
            );

            return result;
        }

        public async Task<int> DeletePatientAsync(int patientId)
        {
            var result = await _context.Database.ExecuteSqlInterpolatedAsync(
                $"EXEC sp_DeletePatient @PatientId={patientId}"
            );

            return result;
        }
    }
}

    STEP 11: Middleware Create

    namespace LoginPatient.Middleware
{
    public class RequestLoggingMiddleware
    {
        private readonly RequestDelegate _next;

        public RequestLoggingMiddleware(RequestDelegate next)
        {
            _next = next;
        }

        public async Task InvokeAsync(HttpContext context)
        {
            Console.WriteLine("========================================");
            Console.WriteLine("Request Started");
            Console.WriteLine($"Method: {context.Request.Method}");
            Console.WriteLine($"Path: {context.Request.Path}");
            Console.WriteLine($"Time: {DateTime.Now}");

            await _next(context);

            Console.WriteLine($"Response Status: {context.Response.StatusCode}");
            Console.WriteLine("Request Completed");
            Console.WriteLine("========================================");
        }
    }
}

    STEP 12: Controller Create

    Right-click on Controllers folder → Add → Controller → API Controller - Empty → Name: PatientController.cs

    using Microsoft.AspNetCore.Mvc;
using LoginPatient.Models;
using LoginPatient.Services;

namespace LoginPatient.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class PatientController : ControllerBase
    {
        private readonly IPatientService _service;

        public PatientController(IPatientService service)
        {
            _service = service;
        }

        [HttpGet]
        public async Task<IActionResult> GetAllPatients()
        {
            var result = await _service.GetAllPatientsAsync();
            return Ok(result);
        }

        [HttpGet("{patientId}")]
        public async Task<IActionResult> GetPatientById(int patientId)
        {
            var patient = await _service.GetPatientByIdAsync(patientId);

            if (patient == null)
                return NotFound(new { message = "Patient not found" });

            return Ok(patient);
        }

        [HttpPost]
        public async Task<IActionResult> AddPatient([FromBody] PatientModel patient)
        {
            if (!ModelState.IsValid)
                return BadRequest(ModelState);

            var result = await _service.AddPatientAsync(patient);
            return Ok(new { message = "Patient saved successfully", rowsAffected = result });
        }

        [HttpDelete("{patientId}")]
        public async Task<IActionResult> DeletePatient(int patientId)
        {
            var result = await _service.DeletePatientAsync(patientId);

            if (result == 0)
                return NotFound(new { message = "Patient not found" });

            return Ok(new { message = "Patient deleted successfully" });
        }
    }
}

using LoginPatient.Data;

app.UseCors("AllowAngular");
