    SSMS — Employee Stored Procedures

    Syntax:

    CREATE PROCEDURE procedureName(parameters)

  AS BEGIN
    SQL STATEMENTS

  END; 

***Syntax to call/invoke ***
EXEC procedureName

    1. Create Employee Table

    CREATE TABLE emp (
    eno INT,
    ename  VARCHAR(50),
    salary MONEY
);

    2. Create Procedure to Add Employee

    CREATE PROCEDURE proc_addemp
    @eno INT,
    @ename VARCHAR(50),
    @sal MONEY
AS
BEGIN
    INSERT INTO emp
    VALUES
    (
        @eno,
        @ename,
        @sal
    );
END;

    To Execute:

    EXEC proc_addemp;

    To Insert Values:

    EXEC proc_addemp 101, 'Anil', 2000;

    3. Create Procedure to Update Employee Details

    Q) Create a procedure to update employee details based on eno?

    CREATE PROCEDURE proc_updateemp
    @eno INT,
    @ename VARCHAR(50),
    @sal MONEY
AS
BEGIN
    UPDATE emp
    SET
        ename = @ename,
        salary = @sal
    WHERE eno = @eno;
END;

    To Execute:

    EXEC proc_updateemp;

    To Update Values:

    EXEC proc_updateemp 101, 'anilkumar', 30000;

    4. Create Procedure to Display Employee Details

    Q) Create a procedure to display employee details?

    CREATE PROCEDURE proc_getemp
AS
BEGIN
    SELECT *
    FROM emp;
END;

    To Execute:

    EXEC proc_getemp;

    5. Create Procedure to Delete Employee Details

    Q) Create a procedure to delete employee details based on employee number?

    CREATE PROCEDURE proc_deleteemp
    @eno INT
AS
BEGIN
    DELETE FROM emp
    WHERE eno = @eno;
END;

    To Execute:

    EXEC proc_deleteemp;

    To Delete Based on Employee ID:

    EXEC proc_deleteemp 101;

│ proc_updateemp               │ → UPDATE
│ proc_getemp                  │ → SELECT
│ proc_deleteemp               │ → DELETE
└──────────────────────────────┘

    Employee CRUD — ASP.NET Core Web API

    1. EmployeeModel.cs

    using System.ComponentModel.DataAnnotations;

namespace Office.Models
{
    public class EmployeeModel
    {
        [Key]
        public int EmpId { get; set; }

        public string EmpFName { get; set; }

        public string EmpLName { get; set; }

        public string Gender { get; set; }
    }
}

    2. OfficeDbContext.cs

    using Microsoft.EntityFrameworkCore;
using Office.Models;

namespace Office.Data
{
    public class OfficeDbContext : DbContext
    {
        public OfficeDbContext(DbContextOptions<OfficeDbContext> options)
            : base(options)
        {

        }

        public DbSet<EmployeeModel> Employee { get; set; }

        // Get Employees
        public List<EmployeeModel> GetEmployees()
        {
            return Employee.ToList();
        }

        // Create / Update Employee
        public string CreateEmployee(EmployeeModel employee)
        {
            var result = this.Database.ExecuteSqlRaw(
                "EXEC InsertEmployee @EmpId={0}, @EmpFName={1}, @EmpLName={2}, @Gender={3}",
                employee.EmpId,
                employee.EmpFName,
                employee.EmpLName,
                employee.Gender
            );

            return result.ToString();
        }

        // Delete Employee
        public string DeleteEmployee(int empId)
        {
            var result = this.Database.ExecuteSqlRaw(
                "EXEC DeleteEmployee @EmpId={0}",
                empId
            );

            return result.ToString();
        }
    }
}

    5. EmployeeController.cs

    using Microsoft.AspNetCore.Mvc;
using Office.Data;
using Office.Models;

namespace Office.Controllers
{
    [Route("[controller]")]
    [ApiController]
    public class EmployeeController : ControllerBase
    {
        private readonly OfficeDbContext _context;

        public EmployeeController(OfficeDbContext context)
        {
            _context = context;
        }

        [HttpGet]
        public List<EmployeeModel> GetEmployees()
        {
            return _context.GetEmployees();
        }

        [HttpPost]
        public IActionResult CreateEmployee(EmployeeModel employee)
        {
            string str = _context.CreateEmployee(employee);

            return Ok(new JsonResult(str));
        }

        [HttpDelete("{empId}")]
        public IActionResult DeleteEmployee(int empId)
        {
            string str = _context.DeleteEmployee(empId);

            return Ok(new JsonResult(str));
        }
    }
}
