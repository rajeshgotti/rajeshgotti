    parent.ts
    import { CommonModule } from '@angular/common';
import { Component, OnInit ,ChangeDetectionStrategy} from '@angular/core';
import { FormControl, Validators } from '@angular/forms';
import { FormGroup, ReactiveFormsModule } from '@angular/forms';
import { MatCheckboxModule } from '@angular/material/checkbox';
import { MatFormFieldModule } from '@angular/material/form-field';
import { MatInputModule } from '@angular/material/input';
import { MatRadioModule } from '@angular/material/radio';
import { MatSelectModule } from '@angular/material/select';
import {MatDatepickerModule} from '@angular/material/datepicker';
import {provideNativeDateAdapter} from '@angular/material/core';
import { Child } from '../child/child';
import { MatTimepickerModule } from '@angular/material/timepicker';

export interface DataFromParent {
  firstNameFc: string;
  lastNameFc: string;
  genderFc: string;
  dobFc: Date;
  cityFc: number;
  emailFc: string;
  ageFc: number;
  passwordFc: string;
  addressFc: string;
  appointmentTimeFc: string;
  cricketFc: boolean;
  footballFc: boolean;
  carromFc: boolean;
  skillsFc: string[];
  resumeFileName?: string;
}

@Component({
  selector: 'app-parent',
  imports: [Child,MatDatepickerModule,CommonModule,MatTimepickerModule,MatFormFieldModule,MatInputModule,
  MatRadioModule,MatCheckboxModule,ReactiveFormsModule,MatSelectModule],
  changeDetection: ChangeDetectionStrategy.OnPush,
  providers: [provideNativeDateAdapter()],
  templateUrl: './parent.html',
  styleUrl: './parent.css'
})
export class Parent implements OnInit{

  myForm:FormGroup
  city:any[]=[]
  skills: string[] = ['Angular', 'React', 'Vue', 'Node.js', 'Python'];

  ngOnInit(): void {
    this.getCities();
    this.myForm=new FormGroup({
     firstNameFc:new FormControl('',Validators.required),
     lastNameFc:new FormControl(''),
     genderFc:new FormControl(''),
     cityFc:new FormControl(''),
     carromFc:new FormControl(''),
     footballFc:new FormControl(''),
     cricketFc:new FormControl(''),
     dobFc:new FormControl(''),
     addressFc: new FormControl(''),                  // Textarea
     appointmentTimeFc: new FormControl(''),          // Time picker
     emailFc: new FormControl('', Validators.email),  // Email
     ageFc: new FormControl(null),                    // Number
     passwordFc: new FormControl(''),                 // Password
     resumeFc: new FormControl(null), 
     skillsFc: new FormControl([], Validators.required),
    })
  }
  getCities():void{
    this.city=[
      {cityId:101,cityName:'Hyderabad'},
      {cityId:102,cityName:'VijayaWada'},
      {cityId:103,cityName:'Maharastra'},
      {cityId:104,cityName:'Bengaloor'}
    ]
  }
  sendDataToChild:any;
  data:DataFromParent[]=[]
  submitForm(){
    if (this.myForm.valid) {
      const formValue = this.myForm.value as DataFromParent;
      this.data.push(formValue);
      console.log(this.data, 'array');
    }
  }
  receiveDataFromChild(data: DataFromParent[]): void {
    const item = data[0];
    this.myForm.patchValue({
      firstNameFc: item.firstNameFc,
      lastNameFc: item.lastNameFc,
      genderFc: item.genderFc,
      dobFc: item.dobFc,
      cityFc: item.cityFc,
      emailFc: item.emailFc,
      ageFc: item.ageFc,
      passwordFc: item.passwordFc,
      addressFc: item.addressFc,
      appointmentTimeFc: item.appointmentTimeFc,
      cricketFc: item.cricketFc,
      footballFc: item.footballFc,
      carromFc: item.carromFc,
      skillsFc: item.skillsFc,
    });
  }

  resetForm(): void {
    this.myForm.reset();
    this.data = [];
  }  

  onFileSelected(event: any) {
    const file = event.target.files[0];
    if (file) {
      this.myForm.patchValue({
        resumeFc: file,
      });
      // Store just the filename
      const latest = this.data[this.data.length - 1];
      if (latest) {
        latest.resumeFileName = file.name;
      }
    }
  }

onSelectall(event: any) {
    if (event.checked) {
      this.myForm.patchValue({ skillsFc: this.skills });
    } else {
      this.myForm.patchValue({ skillsFc: [] });
    }
  }
}

    child.ts
    import { CommonModule } from '@angular/common';
import{Component,effect,EventEmitter,Input,OnChanges,OnInit,Output,SimpleChange,SimpleChanges}from'@angular/core';
import { DataFromParent } from '../parent/parent';
import { SignalDemo } from '../../signal-demo';
import { AgGridModule } from 'ag-grid-angular';

@Component({
  selector: 'app-child',
  imports: [CommonModule,AgGridModule],
  templateUrl: './child.html',
  styleUrl: './child.css'
})
export class Child implements OnChanges{

  @Input() dataFromParent: DataFromParent[] = []; 
  @Output() sendDataToParent=new EventEmitter<DataFromParent[]>();

  ngOnChanges(changes: SimpleChanges): void {
    if (changes['dataFromParent'] && changes['dataFromParent'].currentValue) {
      console.log('Data received from parent:', this.dataFromParent);
    }
  }
  edit(i){
    const dataToEdit = this.dataFromParent[i];
    console.log('Data to edit:', dataToEdit);
    this.sendDataToParent.emit([this.dataFromParent[i]]);
  }
  delete(i){
    this.dataFromParent.splice(i, 1);
  }

  onGridReady(params: any) {
    params.api.sizeColumnsToFit();
  }
}
