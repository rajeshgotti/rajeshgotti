    example:2 If -- Else_if -- if

        <!-- IF value === 'a' -->
<div *ngIf="regForm.get('firstNameFc')?.value === 'a'; else checkB">
  A value
</div>

<!-- ELSE IF value === 'b' -->
<ng-template #checkB>
  <div *ngIf="regForm.get('firstNameFc')?.value === 'b'; else checkC">
    B value
  </div>
</ng-template>

<!-- ELSE -->
<ng-template #checkC>
  <div>
    C value
  </div>
</ng-template>
