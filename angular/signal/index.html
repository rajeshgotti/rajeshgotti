    Signal & BehaviourSubject

    demoService.ts
    //->we created service filed wihtName demoService.ts file
import { Injectable, signal } from '@angular/core';
import { BehaviorSubject } from 'rxjs';

@Injectable({
  providedIn: 'root'
})
export class SignalDemo {
  //signal
  currentMedSignal=signal<any>(null); //creating signal

  currentMedData(data: any) {
     this.currentMedSignal.set(data);
     }  

  //behaviorSubject
  private vitalsUnitsStatus = new BehaviorSubject<any>(false);
  vitalsUnitsStatus$ = this.vitalsUnitsStatus.asObservable(); 

  storevitalsUnitsStatus(value){
    this.vitalsUnitsStatus.next(value)
   }
}
    parent.ts
    export class Parent implements OnInit{

constructor(private ser:SignalDemo) {}

 getCities():void{
    this.city=[
      {cityId:101,cityName:'Hyderabad'},
      {cityId:102,cityName:'VijayaWada'},
      {cityId:103,cityName:'Maharastra'},
      {cityId:104,cityName:'Bengaloor'}
    ]
    this.ser.currentMedData(this.city); // Set the signal with the city data
    this.ser.storevitalsUnitsStatus(this.city); // Store the city data in BehaviorSubject
  }
}
    child.ts
    export class Child implements OnChanges{
  constructor(private ser:SignalDemo){
    // using signal
    effect(() => {
      let transmitData=this.ser.currentMedSignal()
          console.log(transmitData,'data using signal'); 
      }) ,
      // using BehaviorSubject
     this.ser.vitalsUnitsStatus$.subscribe((data) => {
        console.log(data, 'data using BehaviorSubject');
      }); 

  }
}
