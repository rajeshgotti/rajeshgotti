    Step 3: Create Controller Class (API Layer)

    using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using StudenCrud.Data;
using StudenCrud.Model;

namespace StudenCrud.Controllers
{
    [ApiController]
    [Route("api/[controller]")]
    public class StudentsController : ControllerBase
    {
        private readonly StudentDbContext _context;

        public StudentsController(StudentDbContext context)
        {
            _context = context;
        }

        // GET: All Students
        [HttpGet]
        public async Task<ActionResult<IEnumerable<StudentModel>>> GetStudents()
        {
            return await _context.Students.ToListAsync();
        }

        // GET: Student by Id
        [HttpGet("{id}")]
        public async Task<ActionResult<StudentModel>> GetStudent(int id)
        {
            var student = await _context.Students.FindAsync(id);

            if (student == null)
                return NotFound();

            return student;
        }

        // POST: Insert / Update Student
        [HttpPost]
        public async Task<ActionResult<StudentModel>> SaveStudent(StudentModel student)
        {
            if (student.Id == 0)
            {
                _context.Students.Add(student);
            }
            else
            {
                var existingStudent = await _context.Students.FindAsync(student.Id);

                if (existingStudent == null)
                    return NotFound();

                existingStudent.Name = student.Name;
                existingStudent.Age = student.Age;
                existingStudent.Email = student.Email;
                existingStudent.GenderCode = student.GenderCode;
                existingStudent.Hl7Code = student.Hl7Code;
            }

            await _context.SaveChangesAsync();
            return Ok(student);
        }

        // PUT: Update Student
        [HttpPut("{id}")]
        public async Task<IActionResult> UpdateStudent(int id, StudentModel student)
        {
            if (id != student.Id)
                return BadRequest();

            _context.Entry(student).State = EntityState.Modified;

            await _context.SaveChangesAsync();

            return NoContent();
        }

        // DELETE: Remove Student
        [HttpDelete("{id}")]
        public async Task<IActionResult> DeleteStudent(int id)
        {
            var student = await _context.Students.FindAsync(id);

            if (student == null)
                return NotFound();

            _context.Students.Remove(student);
            await _context.SaveChangesAsync();

            return NoContent();
        }

        // GET: Gender Master Data
        [HttpGet("genderv1")]
        public ActionResult<IEnumerable<GenderModel>> GetGenders()
        {
            var genders = new List<GenderModel>
            {
                new GenderModel { AdministrativeSexId = 1, ConceptCode = "A", ConceptName = "Ambiguous", PreferredConceptName = "Ambiguous", CodeSystemOID = "2.16.840.1.113883.12.1", Hl7Code = "HL70001" },
                new GenderModel { AdministrativeSexId = 2, ConceptCode = "F", ConceptName = "Female", PreferredConceptName = "Female", CodeSystemOID = "2.16.840.1.113883.12.1", Hl7Code = "HL70001" },
                new GenderModel { AdministrativeSexId = 3, ConceptCode = "M", ConceptName = "Male", PreferredConceptName = "Male", CodeSystemOID = "2.16.840.1.113883.12.1", Hl7Code = "HL70001" }
            };

            return Ok(genders);
        }

        // GET: Marital Status Data
        [HttpGet("marital")]
        public ActionResult<IEnumerable<MaritalModel>> Marital()
        {
            var marital = new List<MaritalModel>
            {
                new MaritalModel { MaritalStatusId = 3, ConceptCode = "D", ConceptName = "Divorced", PreferredConceptName = "Divorced", Hl7Code = "HL70002" },

        }
    }

    Step 4: Configure DbContext & Services (Program.cs)

    using Microsoft.EntityFrameworkCore;
using StudenCrud.Data;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddControllers();
builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

// DbContext Registration
builder.Services.AddDbContext<StudentDbContext>(options =>
    options.UseSqlServer(builder.Configuration.GetConnectionString("DefaultConnection")));

// CORS
builder.Services.AddCors(options =>
{
    options.AddPolicy("AllowAngularClient",
        policy =>
        {
            policy.WithOrigins("http://localhost:4200")
                  .AllowAnyHeader()
                  .AllowAnyMethod();
        });
});

var app = builder.Build();

if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseHttpsRedirection();
app.UseCors("AllowAngularClient");
app.UseAuthorization();

app.MapControllers();

app.Run();

import { HttpClient } from '@angular/common/http';
import { Injectable } from '@angular/core';
import { environment } from './environment';

@Injectable({
  providedIn: 'root'
})
export class StudentService {

  constructor(private http: HttpClient) {}

  getStudent() {
    return this.http.get(environment.baseUrl + environment.Student);
  }

  getgender() {
    return this.http.get(environment.baseUrl + environment.gender);
  }

  maritalstatus() {
    return this.http.get(environment.baseUrl + environment.maritalstatus);
  }

  postStudent(body: any) {
    return this.http.post(environment.baseUrl + environment.Student, body);
  }

  del(id: number) {
    return this.http.delete(environment.baseUrl + environment.Student + '/' + id);
  }
}

import { Component, OnInit, OnDestroy } from '@angular/core';
import { StudentService } from '../student-service';
import { Subject } from 'rxjs';
import { FormControl, FormGroup } from '@angular/forms';
import { StudentModel } from './studentModel';

@Component({
  selector: 'app-student',
  standalone: true
})
export class Student implements OnInit, OnDestroy {

  private destroy$ = new Subject<void>();

  studentForm!: FormGroup;

  genders: any[] = [];
  maritalList: any[] = [];
  getStudentData: any;

  studentid: number | null = null;

  constructor(private ser: StudentService) {}

  ngOnInit() {

    this.studentForm = new FormGroup({
      firstName: new FormControl(''),
      age: new FormControl(),
      email: new FormControl(),
      gender: new FormControl(),
      marital: new FormControl()
    });

    this.getStudent();
    this.gender();
    this.maritalstatu();
  }

  getStudent() {
    this.ser.getStudent().subscribe({
      next: (res) => {
        this.getStudentData = res;
      }
    });
  }

  gender() {
    this.ser.getgender().subscribe({
      next: (res) => {
        this.genders = res as any[];
      }
    });
  }

  maritalstatu() {
    this.ser.maritalstatus().subscribe({
      next: (res) => {
        this.maritalList = res as any[];
      }
    });
  }

  edit(item: any) {
    this.studentid = item?.id;

    let genobj = this.genders.find(ele => ele.conceptCode === item.genderCode);
    let maritalobj = this.maritalList.find(ele => ele.hl7Code === item.hl7Code);

    this.studentForm.patchValue({
      firstName: item?.name,
      age: item?.age,
      email: item?.email,
      gender: genobj?.conceptCode,
      marital: maritalobj?.hl7Code
    });
  }

  saveStudent() {

    let stdobj = new StudentModel();

    stdobj.id = this.studentid ?? 0;
    stdobj.name = this.studentForm.get('firstName')?.value;
    stdobj.age = this.studentForm.get('age')?.value;
    stdobj.email = this.studentForm.get('email')?.value;

    stdobj.genderCode = this.studentForm.get('gender')?.value;
    stdobj.hl7Code = this.studentForm.get('marital')?.value;

    this.ser.postStudent(stdobj).subscribe({
      next: () => {
        alert('Successfully saved');
        this.studentForm.reset();
        this.getStudent();
      }
    });
  }

  del(item: any) {
    this.ser.del(item?.id).subscribe({
      next: () => {
        alert('Deleted');
        this.getStudent();
      }
    });
  }

  clear() {
    this.studentForm.reset();
    this.studentid = null;
  }

  ngOnDestroy() {
    this.destroy$.next();
    this.destroy$.complete();
  }
}
