    ts.file
    import { Component } from '@angular/core';

@Component({
  selector: 'app-parent',
  templateUrl: './parent.component.html',
  styleUrls: ['./parent.component.css']
})
export class ParentComponent {
  cities = [
    { cityId: 101, cityName: 'Hyderabad' },
    { cityId: 102, cityName: 'VijayaWada' },
    { cityId: 103, cityName: 'Maharashtra' },
    { cityId: 104, cityName: 'Bengaluru' }
  ];

  skills: string[] = ['Angular', 'React', 'Vue', 'Node.js', 'Python'];

  formModel: any = {
    firstName: '',
    lastName: '',
    gender: '',
    dob: '',
    city: '',
    email: '',
    age: null,
    password: '',
    address: '',
    appointmentTime: '',
    cricket: false,
    football: false,
    carrom: false,
    skills: [],
    resume: null
  };

  fileName = '';

  onSubmit(form: any): void {
    if (form.valid) {
      console.log('Form Submitted:', this.formModel);
    } else {
      console.warn('Form Invalid');
    }
  }

  onFileChange(event: any): void {
    const file = event.target.files[0];
    if (file) {
      this.formModel.resume = file;
      this.fileName = file.name;
    }
  }

  onSelectAllSkills(event: any): void {
    if (event.target.checked) {
      this.formModel.skills = [...this.skills];
    } else {
      this.formModel.skills = [];
    }
  }
}
