    open sidenav when click on Button

    Open sideNav when click on Button

    component.ts

    import { ViewChild } from '@angular/core';
import { MatSidenav, MatSidenavModule } from '@angular/material/sidenav';

imports: [MatSidenavModule]

@ViewChild('demo')
demosidebar!: MatSidenav;

openSideNav() {
  this.demosidebar.open();
}

close() {
  this.demosidebar.close();
}

    Explanation

    • @ViewChild('demo') use chesi HTML lo unna mat-sidenav ni TypeScript lo access chestunnam.
    • open() method sidenav ni open chestundi.
    • close() method sidenav ni close chestundi.

    component.html

    <button mat-raised-button color="primary" (click)="openSideNav()">Open Button</button>
