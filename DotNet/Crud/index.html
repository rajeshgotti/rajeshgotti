    HospitalController.cs
    using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using WebApplication4.Models;

namespace WebApplication4.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class HospitalController : ControllerBase
    {
        private readonly HospitalDbContext _context;

        public HospitalController(HospitalDbContext context)
        {
            _context = context;
        }

        // ✅ GET all hospitals with their addresses
        [HttpGet]
        public async Task<ActionResult<IEnumerable<Hospital>>> GetHospitals()
        {
            return await _context.Hospitals
                .Include(h => h.Addresses) // Eager load addresses
                .ToListAsync();
        }

        // ✅ GET hospital by Id with addresses
        [HttpGet("{id}")]
        public async Task<ActionResult<Hospital>> GetHospital(int id)
        {
            var hospital = await _context.Hospitals
                .Include(h => h.Addresses) // Eager load addresses
                .FirstOrDefaultAsync(h => h.Id == id);

            if (hospital == null)
            {
                return NotFound();
            }

            return hospital;
        }

        // ✅ POST - Add a new hospital with addresses
        [HttpPost]
        public async Task<ActionResult<Hospital>> PostHospital(Hospital hospital)
        {
            _context.Hospitals.Add(hospital);
            await _context.SaveChangesAsync();

            return CreatedAtAction(nameof(GetHospital), new { id = hospital.Id }, hospital);
        }

        [HttpPut("{id}")]
        public async Task<IActionResult> PutHospital(int id, Hospital hospital)
        {
            if (id != hospital.Id) return BadRequest();

            _context.Entry(hospital).State = EntityState.Modified;

            // Update related addresses
            var existingHospital = await _context.Hospitals
                .Include(h => h.Addresses)
                .FirstOrDefaultAsync(h => h.Id == id);
            if (existingHospital != null)
            {
                _context.Addresses.RemoveRange(existingHospital.Addresses);
                existingHospital.Addresses = hospital.Addresses;
            }

            try
            {
                await _context.SaveChangesAsync();
            }
            catch (DbUpdateConcurrencyException)
            {
                if (!_context.Hospitals.Any(e => e.Id == id))
                {
                    return NotFound();
                }
                else
                {
                    throw;
                }
            }

            return NoContent();
        }

        // ✅ DELETE - Remove hospital and its addresses
        [HttpDelete("{id}")]
        public async Task<IActionResult> DeleteHospital(int id)
        {
            var hospital = await _context.Hospitals
                .Include(h => h.Addresses)
                .FirstOrDefaultAsync(h => h.Id == id);
            if (hospital == null)
            {
                return NotFound();
            }

            _context.Hospitals.Remove(hospital);
            await _context.SaveChangesAsync();

            return NoContent();
        }
    }
}

using Microsoft.AspNetCore.Http;
using Microsoft.AspNetCore.Mvc;
using StudentWebAPI.Models;

namespace StudentWebAPI.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class StudentController : ControllerBase
    {
        private readonly StudentContext _context;
        public StudentController(StudentContext context)
        {
            _context = context;
        }
        [HttpGet]
        public IActionResult GetStudents()
        {
            var students = _context.Student.ToList();
            return Ok(students);
        }
        [HttpGet("{id}")]
        public IActionResult GetStudent(int id)
        {
            var student = _context.Student.Find(id);
            if (student == null)
            {
                return NotFound("Student not found");
            }
            return Ok(student);
        }
        [HttpPost]
        public IActionResult AddStudent(Student student)
        {
            _context.Student.Add(student);
            _context.SaveChanges();
            return Ok("Student Added");
        }

        [HttpPut("{id}")]

        public IActionResult EditStudent(int id, StudentDto student)
        {
            var existingStudent = _context.Student.Find(id);
            if (existingStudent == null)
            {
                return NotFound("Student not found");
            }
            existingStudent.Name = student.Name;
            existingStudent.Email = student.Email;
            existingStudent.Age = student.Age;
            existingStudent.Gender = student.Gender;
            existingStudent.Password = student.Password;

            _context.SaveChanges();
            return Ok("Student Updated");
        }
        [HttpDelete("{id}")]
        public IActionResult DeleteStudent(int id)
        {
            var student = _context.Student.Find(id);
            if (student == null)
            {
                return NotFound("Student not found");
            }
            _context.Student.Remove(student);
            _context.SaveChanges();
            return Ok("Student Deleted");
        }

    }
}

using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using StudentApi.Models;

namespace StudentApi.Controllers
{
    [ApiController]
    [Route("api/[controller]")]
    public class StudentsController : ControllerBase
    {
        private readonly StudentDbContext _context;

        public StudentsController(StudentDbContext context)
        {
            _context = context;
        }

        // GET: api/students (Get All via Stored Procedure)
        [HttpGet]
        public async Task<ActionResult<IEnumerable<StudentModel>>> GetStudents()
        {
            var students = await _context.Students
                .FromSqlInterpolated($"EXEC GetAllStudents")
                .ToListAsync();

            return students;
        }

        // GET: api/students/5 (Get By Id via Stored Procedure)
        [HttpGet("{id}")]
        public async Task<ActionResult<StudentModel>> GetStudent(int id)
        {
            var student = (await _context.Students
                .FromSqlInterpolated($"EXEC dbo.GetStudentById @Id={id}")
                .ToListAsync())
                .FirstOrDefault();

            if (student == null)
                return NotFound();

            return student;
        }
        // POST: api/students (Insert via Stored Procedure)
        [HttpPost]
        public async Task<IActionResult> CreateStudent(StudentModel student)
        {
            await _context.Database.ExecuteSqlInterpolatedAsync(
                $"EXEC AddStudent @Name={student.Name}, @Age={student.Age}, @Email={student.Email}"
            );

            return Ok("Student inserted successfully.");
        }

        // PUT: api/students/5 (Update via Stored Procedure)
        [HttpPut("{id}")]
        public async Task<IActionResult> UpdateStudent(int id, StudentModel student)
        {
            // Use 'id' from path, ignore student.Id
            await _context.Database.ExecuteSqlInterpolatedAsync(
                $"EXEC UpdateStudent @Id={id}, @Name={student.Name}, @Age={student.Age}, @Email={student.Email}"
            );

            return Ok("Student updated successfully.");
        }

        // DELETE: api/students/5 (Delete via Stored Procedure)
        [HttpDelete("{id}")]
        public async Task<IActionResult> DeleteStudent(int id)
        {
            await _context.Database.ExecuteSqlInterpolatedAsync(
                $"EXEC DeleteStudent @Id={id}"
            );

            return Ok("Student deleted successfully.");
        }

        // GET: api/students/search/{keyword} (Search via Stored Procedure)
        [HttpGet("search/{keyword}")]
        public async Task<ActionResult<IEnumerable<StudentModel>>> SearchStudents(string keyword)
        {
            var students = await _context.Students
                .FromSqlInterpolated($"EXEC SearchStudents @Keyword={keyword}")
                .ToListAsync();

            return students;
        }

        // GET: api/students/sort/{sortBy}/{sortDirection} (Sort via Stored Procedure)
        [HttpGet("sort/{sortBy}/{sortDirection}")]
        public async Task<ActionResult<IEnumerable<StudentModel>>> SortStudents(string sortBy,
         string sortDirection)
        {
            var students = await _context.Students
                .FromSqlInterpolated($"EXEC SortStudents @SortBy={sortBy}, @SortDirection={sortDirection}")
                .ToListAsync();

            return students;
        }
    }
}
