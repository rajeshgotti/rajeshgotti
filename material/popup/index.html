    Angular Material Dialog Using Template-Driven Popup

    component.html

    <ng-template #moveRef>
  <div class="modalHeader">
    <div class="modal-about">
      <div class="font-weight-bold">
        <span class="mx-4">Patient ID : <strong>{{ patientId }}</strong></span>
        <span>Patient Name : <strong>{{ patientName }}</strong></span>

        <button mat-icon-button (click)="closePopup()" aria-label="Close Dialog">
          <mat-icon>close</mat-icon>
        </button>
      </div>
    </div>
  </div>
</ng-template>

    Note: Import MatDialog, MatDialogRef, ViewChild, TemplateRef, and inject.

    import { ViewChild, TemplateRef, inject } from '@angular/core';
import { MatDialog, MatDialogRef } from '@angular/material/dialog';

    constructor(private dialog: MatDialog) {}

    @ViewChild('moveRef')
moveRef!: TemplateRef<any>;

    Explanation

    @ViewChild() use chesi HTML lo unna ng-template ni TypeScript file lo access chestunnam.

    moveRefs = inject(MatDialogRef<any>);

movedropDown() {
  this.moveRefs = this.dialog.open(this.moveRef, {
    width: '100vh',
    height: 'auto'
  });
}

closePopup() {
  this.moveRefs.close();
}

    Explanation

    • dialog.open() use chesi ng-template ni popup ga open chestunnam.
    • close() method use chesi popup ni close chestunnam.

    Angular Material Dialog Using ng-template with Child Component

    component.ts

    Note: Import MatDialog, MatDialogModule, MatDialogRef, TemplateRef, and ViewChild.

    import { Inject, TemplateRef, ViewChild } from '@angular/core';
import { MatDialog, MatDialogModule, MatDialogRef, MAT_DIALOG_DATA } from '@angular/material/dialog';
import { Timeline2Component } from '@shared/components/auditLog-Timeline/timeline2.component';

    imports: [Timeline2Component, MatDialogModule]

    constructor(private dialog: MatDialog, @Inject(MAT_DIALOG_DATA) public data: any) {}

    @ViewChild('auditLogRef')
auditLogRef!: TemplateRef<any>;
auditLogDialogRef!: MatDialogRef<any>;

    openAuditLog(): void {
  this.auditLogDialogRef = this.dialog.open(this.auditLogRef, {
    width: '90%',
    height: '90%',
    disableClose: true,
    panelClass: 'audit-log-dialog'
  });
}

    Explanation

    dialog.open() use chesi ng-template ni popup ga open chestunnam.

    component.html

    <button type="button" mat-icon-button  matTooltip="Audit Log" aria-label="Audit Log"
  (click)="openAuditLog(); $event.stopPropagation()">
  <i class="fa-solid fa-clipboard-list"></i>
</button>

<ng-template #auditLogRef>
  <div class="addContainer">
    <div class="modalHeader">
      <div class="font-weight-bold p-t-5 p-l-10 d-flex flex-wrap align-items-center">
        <span class="me-3">Audit Log</span>
        <span class="ms-3"><span class="text-black">Patient:</span> <span class="text-blue ms-1">{{ patientName }}</span></span>
        <span class="ms-3"><span class="text-black">MRN:</span> <span class="text-blue ms-1">{{ mrn }}</span></span>
        <span class="ms-3"><span class="text-black">DOB:</span> <span class="text-blue ms-1">{{ dob | date:'MM/dd/yyyy' }}</span></span>
        <span class="ms-3"><span class="text-black">Gender:</span> <span class="text-blue ms-1">{{ gender }}</span></span>
      </div>

      <button class="close"  mat-icon-button (click)="auditLogDialogRef.close()"          
             aria-label="Close Dialog">
        <i class="material-icons">close</i>
      </button>

    </div>
    <div class="modal_body mat-dialog-content">
      <app-timeline2
        [patientId]="patientId"
        [visitId]="visitId"
        [patientName]="patientName">
      </app-timeline2>
    </div>
  </div>
</ng-template>

    Explanation

    • ng-template popup lo <app-timeline2> component ni load chestunnam.
    • patientId, visitId, patientName ni @Input() bindings dwara child component ki pass chestunnam.
    • Close icon click cheste auditLogDialogRef.close() popup ni close chestundi.
