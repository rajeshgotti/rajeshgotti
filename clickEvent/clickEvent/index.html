    Difference between (ngModelChange) & (input)

    NOTE
    -> (ngModelChange) gives you the updated bound model value.

-> (input) gives you the raw DOM value via $event.target.value.
    ex: (input)="getSites($event.target.value)"

    const inputValue = (event.target as HTMLInputElement).value;

    if (inputValue?.length > 2) {
      this.historyService.getWsDrugSuggestions(inputValue, 'DRUG', 'F').pipe(takeUntil(this.onDestroy$))
        .subscribe({
          next: (res: any) => {
            this.drugSuggestions  = res?.body || [];

            this.drugSuggestions  = [];
          },
        });
      }else {
      this.drugSuggestions  = [];

    if(!this.pharmacyForm.get('drugName').value){
      this.SearchDrugId='0'

  }
